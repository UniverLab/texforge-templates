\definecolor{editorialink}{HTML}{1D1510}
\definecolor{editorialaccent}{HTML}{87222C}
\definecolor{editorialgold}{HTML}{E6C84A}

  urlcolor=editorialaccent,breaklinks=true,pdfauthor={{{author}}},

% A quiet hierarchy: dark ink for headings, a warm accent for section numbers.
\titleformat{\section}{\normalfont\Large\bfseries\color{editorialink}}
{\textcolor{editorialaccent}{\thesection}}{0.75em}{}
\titlespacing*{\section}{0pt}{1.45\baselineskip}{0.55\baselineskip}
\titleformat{\subsection}{\normalfont\large\bfseries\color{editorialink}}
{\thesubsection}{0.7em}{}
\titlespacing*{\subsection}{0pt}{1.15\baselineskip}{0.4\baselineskip}
\setlength{\emergencystretch}{2em}

% Title block: left aligned, with a gold rule and the author and date in sepia.
\makeatletter
\renewcommand{\maketitle}{\begingroup\raggedright
  {\fontsize{28}{32}\selectfont\color{editorialink}\@title\par}\vspace{8pt}
  {\color{editorialgold}\rule{22mm}{2pt}\par}\vspace{6pt}
  {\itshape\color{editorialmuted}\@author\quad\@date\par}\vspace{14pt}\endgroup}
\makeatother
\renewenvironment{abstract}
{\begin{quote}\small{\scshape\bfseries\color{editorialaccent}\abstractname}\quad\ignorespaces}
  {\end{quote}}

% Replace the three fields below with your own.
\title{{{title}}}
\author{{{author}}}
\date{{{date}}}
