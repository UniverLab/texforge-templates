  \edef\TotB{\fpeval{\QtyB*\PriceB}}
  \edef\TotC{\fpeval{\QtyC*\PriceC}}
  \edef\Sub{\fpeval{\TotA+\TotB+\TotC}}
  \edef\Adv{\fpeval{\Sub*\AdvPct/100}}
  \edef\Bal{\fpeval{\Sub-\Adv}}
  \edef\BalPct{\fpeval{100-\AdvPct}}
