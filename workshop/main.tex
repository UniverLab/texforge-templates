def classify(values):
    even, odd = [], []
    for value in values:
        (even if value % 2 == 0 else odd).append(value)
    return even, odd

def parse_values(text):
    if not text.strip():
        raise ValueError("input is empty")
    return [int(value) for value in text.split(",")]
