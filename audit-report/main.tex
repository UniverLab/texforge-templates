% A finding is one environment; its metadata are keys.
\pgfkeys{/finding/.cd,id/.store in=\fid,title/.store in=\ftitle,
  severity/.store in=\fsev,likelihood/.store in=\flik,impact/.store in=\fimp,
  effort/.store in=\feff,status/.store in=\fstat}
\newcommand{\findingpart}[1]{\par\vspace{4pt}{\sffamily\scriptsize\scshape\bfseries
    \color{auditaccent}#1}\par\nopagebreak\vspace{-3pt}}
\newenvironment{finding}[1][]
{\pgfkeys{/finding/.cd,id=F-00,title=Untitled,severity=info,likelihood=1,
    impact=1,effort=small,status=open,#1}%
  \stepcounter{cnt\fsev}%
  \placeinmatrix{\flik}{\fimp}{\fid}%
  \addcontentsline{toc}{subsection}{\fid\hspace{0.8em}\ftitle}%
  \needspace{9\baselineskip}%
  \par\vspace{10pt}\noindent
  \begin{tcolorbox}[enhanced,colback=benchlamp!35,colframe=benchlamp!35,
      boxrule=0pt,arc=0pt,borderline west={3pt}{0pt}{sev\fsev},
      left=10pt,right=10pt,top=7pt,bottom=7pt,before skip=0pt,after skip=4pt]
    \setlength{\parskip}{0pt}%
    \noindent\parbox[t]{\dimexpr\linewidth-2.4cm\relax}{%
      {\sffamily\bfseries\color{sev\fsev}\fid}\hspace{0.8em}%
      {\fontsize{14}{17}\selectfont\color{auditink}\ftitle}}%
    \hfill\parbox[t]{2.2cm}{\raggedleft\severitybadge{\fsev}}%
    \par\vspace{5pt}
    {\color{auditrule}\hrule height 0.4pt}\par\vspace{5pt}
    \begin{tabular}{@{}p{3.1cm}p{3.1cm}p{3.1cm}p{3.1cm}@{}}
      \metalabel{\lblLikelihood} & \metalabel{\lblImpact} &
      \metalabel{\lblEffort} & \metalabel{\lblFindingStatus} \\[1pt]
      \flik\,/\,5 & \fimp\,/\,5 & \csname lblef\feff\endcsname &
      \csname lblfs\fstat\endcsname \\
    \end{tabular}
  \end{tcolorbox}\nopagebreak}
{\par\vspace{6pt}}

% Risk matrix: likelihood against impact, one coloured cell per pair.
\newcommand{\riskmatrix}{%
  \begin{center}
    \begin{tikzpicture}[x=24mm,y=14mm]
      \foreach \l in {1,...,5}{\foreach \i in {1,...,5}{%
          \pgfmathtruncatemacro{\sc}{\l*\i}%
          \ifnum\sc>14 \def\rc{sevcritical}\else\ifnum\sc>9 \def\rc{sevhigh}\else
          \ifnum\sc>4 \def\rc{sevmedium}\else\def\rc{sevlow}\fi\fi\fi
          \fill[\rc!24] (\i-1,\l-1) rectangle (\i,\l);
          \draw[white,line width=1.4pt] (\i-1,\l-1) rectangle (\i,\l);
          \node[font=\sffamily\bfseries\small,text=\rc!85!black,text width=22mm,align=center]
          at (\i-0.5,\l-0.5) {\matrixcell{\l}{\i}};}}
      \foreach \n in {1,...,5}{%
        \node[font=\sffamily\small,text=auditmuted] at (\n-0.5,-0.4) {\n};
        \node[font=\sffamily\small,text=auditmuted] at (-0.35,\n-0.5) {\n};}
      \node[font=\sffamily\scriptsize\scshape,text=auditmuted] at (2.5,-0.95) {\lblImpact\ $\rightarrow$};
      \node[font=\sffamily\scriptsize\scshape,text=auditmuted,rotate=90] at (-0.85,2.5)
      {\lblLikelihood\ $\rightarrow$};
    \end{tikzpicture}
  \end{center}}
