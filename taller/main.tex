def clasificar(numeros):
    pares, impares = [], []
    for n in numeros:
        if n % 2 == 0:
            pares.append(n)
        else:
            impares.append(n)
    return pares, impares


def validar(entrada):
    if not entrada:
        raise ValueError("entrada vacía")
    return [int(x) for x in entrada.split(",")]
  \end{code}
